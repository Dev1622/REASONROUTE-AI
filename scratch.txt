import { GoogleGenAI, Type } from "@google/genai";
import { OptimizationResult } from "../types";
import { nearestNeighbor, twoOpt, routeDistance, Node } from "./baslineSolver";

const ai = new GoogleGenAI({ apiKey: process.env.API_KEY });

const MODEL_NAME = "gemini-3-flash-preview";

export const optimizeRoute = async (
  stopsInput: string,
  contextImageBase64?: string,
  contextImageMimeType?: string
): Promise<OptimizationResult> => {

  const prompt = `
    You are the ReasonRoute AI, a Gemini Supply Chain Intelligence Agent.
    Your task is to optimize multimodal delivery routes using a Spatial-Reasoning Framework.
    
    Workflow Stages:
    1. Normalization: Parse the input list of stops. Normalize them into a graph.
    2. Baseline Simulation: Generate a greedy, nearest-neighbor route as a baseline.
    3. Pattern Detection (Preprocessor P): Identify Hubs (Parity-Star), Dense Minors (clusters), and Sunflowers (overlapping zones).
    4. Context Override: If an image is provided, analyze it for disruptions (traffic, flood, roadblocks) and reroute around them.
    5. Recursive Refinement: Generate the optimized route.

    Input Stops:
    ${stopsInput}

    ${contextImageBase64 ? 'A context image (heatmap/traffic map) is provided. Treat this as ground truth for obstructions.' : 'No context image provided. rely on structural optimization.'}

    Output Requirements:
    Return a strictly valid JSON object adhering to the schema.
    - Generate synthetic spatial coordinates (x,y between 10 and 90).
    - nodes
    - baselinePath
    - optimizedPath
    - disruptions
    - logs
    - metrics
  `;

  const parts: any[] = [{ text: prompt }];

  if (contextImageBase64 && contextImageMimeType) {
    parts.push({
      inlineData: {
        data: contextImageBase64,
        mimeType: contextImageMimeType,
      },
    });
  }

  // ✅ ORIGINAL SCHEMA UNCHANGED
  const responseSchema = {
    type: Type.OBJECT,
    properties: {
      nodes: {
        type: Type.ARRAY,
        items: {
          type: Type.OBJECT,
          properties: {
            id: { type: Type.STRING },
            x: { type: Type.NUMBER },
            y: { type: Type.NUMBER },
            type: { type: Type.STRING },
            label: { type: Type.STRING },
          },
          required: ["id", "x", "y", "type", "label"]
        }
      },
      baselinePath: {
        type: Type.ARRAY,
        items: { type: Type.STRING }
      },
      optimizedPath: {
        type: Type.ARRAY,
        items: { type: Type.STRING }
      },
      disruptions: {
        type: Type.ARRAY,
        items: {
          type: Type.OBJECT,
          properties: {
            x: { type: Type.NUMBER },
            y: { type: Type.NUMBER },
            radius: { type: Type.NUMBER },
            description: { type: Type.STRING },
          },
          required: ["x", "y", "radius", "description"]
        }
      },
      logs: {
        type: Type.ARRAY,
        items: { type: Type.STRING }
      },
      metrics: {
        type: Type.OBJECT,
        properties: {
          baseline: {
            type: Type.OBJECT,
            properties: {
              distance: { type: Type.NUMBER },
              time: { type: Type.NUMBER },
              emissions: { type: Type.NUMBER },
            },
            required: ["distance", "time", "emissions"]
          },
          optimized: {
            type: Type.OBJECT,
            properties: {
              distance: { type: Type.NUMBER },
              time: { type: Type.NUMBER },
              emissions: { type: Type.NUMBER },
            },
            required: ["distance", "time", "emissions"]
          }
        },
        required: ["baseline", "optimized"]
      }
    },
    required: ["nodes", "baselinePath", "optimizedPath", "logs", "metrics"]
  };

  try {
    const response = await ai.models.generateContent({
      model: MODEL_NAME,
      contents: { parts },
      config: {
        responseMimeType: "application/json",
        responseSchema,
        temperature: 0,
        topK: 1,
        topP: 0.1
      },
    });

    const text = response.text;
    if (!text) throw new Error("No response from Gemini");

    const parsed = JSON.parse(text);

    // -------------------------------------------------
    // BUILD NODE MAP
    // -------------------------------------------------
    const nodeMap = new Map<string, Node>();
    parsed.nodes.forEach((n: Node) => nodeMap.set(n.id, n));

    const depotId = parsed.nodes[0].id;

    // -------------------------------------------------
    // REAL BASELINE
    // -------------------------------------------------
    let realBaseline = nearestNeighbor(parsed.nodes, depotId);
    realBaseline = twoOpt(realBaseline, nodeMap);

    parsed.baselinePath = realBaseline;

    // -------------------------------------------------
    // SHARED SCORING
    // -------------------------------------------------
    const baseDistance = routeDistance(realBaseline, nodeMap);
    const optDistance = routeDistance(parsed.optimizedPath, nodeMap);

    parsed.metrics = {
      baseline: {
        distance: baseDistance,
        time: baseDistance * 1.3,
        emissions: baseDistance * 0.4,
      },
      optimized: {
        distance: optDistance,
        time: optDistance * 1.3,
        emissions: optDistance * 0.4,
      },
    };

    return parsed as OptimizationResult;

  } catch (error) {
    console.error("Gemini optimization failed:", error);
    throw error;
  }
};

// reasonRouteEngine.ts
// GNOPT-Inspired Spatial Reasoning Engine v4.0
// Pure deterministic math — no AI calls here.
// Responsibilities:
//   Stage 0 — Normalization (constraint arity reduction)
//   Stage 1 — Baseline (composite-cost nearest-neighbor + 2-opt)
//   Stage 2 — Pattern Compression (Hub, Sunflower, Dense-Cluster detection)
//   Stage 3 — Recursive Refinement (spectral partition + local solve)
//   Output  — Structured result with REAL metrics on a single coordinate system

// Public types

export type ConstraintType =
  | "timeWindow"
  | "restrictedZone"
  | "driverShift"
  | "capacity"
  | "exclusion";

export type Constraint = {
  id: string;
  type: ConstraintType;
  details: string;
  appliesTo: string[];
  arity: number;
};

export type EngineInput = {
  nodes: Node[];
  depotId: string;
  constraints?: Constraint[];
  trafficHeatmap?: number[][];
  disruptionZones?: { x: number; y: number; radius: number; description: string }[];
  preferences?: {
    weightTime: number;
    weightEmissions: number;
    weightDistance: number;
  };
};

export type CompressionPattern = {
  type: "hub" | "sunflower" | "denseCluster";
  id: string;
  involvedNodeIds: string[];
  macroNodeId: string;
  description: string;
};

export type EngineResult = {
  baselinePath: string[];
  optimizedPath: string[];
  metrics: {
    baseline: PathMetrics;
    optimized: PathMetrics;
    improvement: {

      timePct: number;
      emissionsPct: number;

  engineLogs: string[];
  recursionDepth: number;
};

// ─────────────────────────────────────────────────────────
// Internal types
// ─────────────────────────────────────────────────────────

type InteractionGraph = {
  nodes: Node[];
  constraints: Constraint[];
};

type Gadget = {
  type: "hub" | "sunflower" | "denseCluster";
  id: string;
  replacedNodeIds: string[];
  macroNode: Node;
  description: string;
};

// ─────────────────────────────────────────────────────────
// Stage 0: Normalization
// ─────────────────────────────────────────────────────────

function normalizeConstraints(
  nodes: Node[],
  constraints: Constraint[],
  logs: string[]
): InteractionGraph {
  const result: Constraint[] = [];
  let helperCount = 0;

  for (const c of constraints) {
    if (c.appliesTo.length <= 3) {
      result.push({ ...c, arity: c.appliesTo.length });
      continue;
    }
    let remaining = [...c.appliesTo];
    while (remaining.length > 2) {
      const chunk = remaining.splice(0, 2);
      const helperName = `_h${helperCount++}`;
      result.push({
        id: `${c.id}_s${helperCount}`,
        type: c.type,
        details: `${c.details} [norm]`,
        appliesTo: [...chunk, helperName],
        arity: 3,
      });
      remaining.unshift(helperName);
    }
    logs.push(`[Norm] Decomposed ${c.id} (arity ${c.appliesTo.length}) into binary tree.`);
  }

  return { nodes, constraints: result };
}

// ─────────────────────────────────────────────────────────
// Stage 2: Pattern Compression
// ─────────────────────────────────────────────────────────

function buildConstraintDegreeMap(g: InteractionGraph): Map<string, number> {
  const map = new Map<string, number>();
  for (const c of g.constraints) {
    for (const nodeId of c.appliesTo) {
      map.set(nodeId, (map.get(nodeId) ?? 0) + 1);
    }
  }
  return map;
}

function getCentroid(nodeIds: string[], allNodes: Node[]): { x: number; y: number } {
  const members = allNodes.filter(n => nodeIds.includes(n.id));
  if (members.length === 0) return { x: 50, y: 50 };
  const x = members.reduce((s, n) => s + n.x, 0) / members.length;
  const y = members.reduce((s, n) => s + n.y, 0) / members.length;
  return { x, y };
}

function detectGeoClusters(
  nodes: Node[],
  threshold: number
): Map<string, string[]> {
  const visited = new Set<string>();
  const clusters = new Map<string, string[]>();

  for (const pivot of nodes) {
    if (visited.has(pivot.id)) continue;
    const members = nodes.filter(n => {
      if (visited.has(n.id)) return false;
      const dx = n.x - pivot.x;
      const dy = n.y - pivot.y;
      return Math.sqrt(dx * dx + dy * dy) <= threshold;
    });
    if (members.length >= 3) {
      members.forEach(m => visited.add(m.id));
      clusters.set(pivot.id, members.map(m => m.id));
    }
  }
  return clusters;
}

function applyGadget(g: InteractionGraph, gadget: Gadget): InteractionGraph {
  const nodes = [
    ...g.nodes.filter(n => !gadget.replacedNodeIds.includes(n.id)),
    gadget.macroNode,
  ];
  const constraints = g.constraints.map(c => ({
    ...c,
    appliesTo: c.appliesTo.map(id =>
      gadget.replacedNodeIds.includes(id) ? gadget.macroNode.id : id
    ),
  }));
  return { nodes, constraints };
}

function runPreprocessorP(
  graph: InteractionGraph,
  depotId: string,
  logs: string[]
): { compressedGraph: InteractionGraph; gadgets: Gadget[]; patterns: CompressionPattern[] } {
  let current = { nodes: [...graph.nodes], constraints: [...graph.constraints] };
  const gadgets: Gadget[] = [];
  const patterns: CompressionPattern[] = [];
  let counter = 0;

  // A. Hub detection (high constraint-degree nodes)
  if (current.constraints.length > 0) {
    const degreeMap = buildConstraintDegreeMap(current);
    for (const [nodeId, degree] of degreeMap) {
      if (degree > 4 && nodeId !== depotId) {
        const hubNode = current.nodes.find(n => n.id === nodeId);
        if (!hubNode) continue;
        const macroId = `HUB_${nodeId}`;
        const macro: Node = { ...hubNode, id: macroId, type: "virtual" };
        const gadget: Gadget = {
          type: "hub",
          id: `g_hub_${counter++}`,
          replacedNodeIds: [nodeId],
          macroNode: macro,
          description: `Hub at ${nodeId} (degree ${degree})`,
        };
        gadgets.push(gadget);
        patterns.push({
          type: "hub",
          id: gadget.id,
          involvedNodeIds: [nodeId],
          macroNodeId: macroId,
          description: `Parity-star hub at node ${nodeId} (${degree} constraint connections). Compressed.`,
        });
        current = applyGadget(current, gadget);
        logs.push(`[Stage 2] Hub detected at ${nodeId} (degree=${degree}). → ${macroId}`);
      }
    }
  }

  // B. Sunflower detection (constraint touching many nodes)
  for (const c of current.constraints) {
    const active = c.appliesTo.filter(id => current.nodes.some(n => n.id === id));
    if (active.length >= 4) {
      const petals = active.slice(1);
      const safe = petals.filter(id => id !== depotId);
      if (safe.length < 2) continue;
      const macroId = `SUN_${c.id}`;
      const { x, y } = getCentroid(active, current.nodes);
      const macro: Node = { id: macroId, x, y, type: "virtual" };
      const gadget: Gadget = {
        type: "sunflower",
        id: `g_sun_${counter++}`,
        replacedNodeIds: safe,
        macroNode: macro,
        description: `Sunflower on ${c.id} (${safe.length} petals)`,
      };
      gadgets.push(gadget);
      patterns.push({
        type: "sunflower",
        id: gadget.id,
        involvedNodeIds: active,
        macroNodeId: macroId,
        description: `Sunflower pattern: constraint ${c.id} links ${active.length} nodes. Petals compressed.`,
      });
      current = applyGadget(current, gadget);
      logs.push(`[Stage 2] Sunflower on ${c.id} (${safe.length} petals) → ${macroId}`);
    }
  }

  // C. Dense geographic cluster detection
  const nonDepotReal = current.nodes.filter(
    n => n.id !== depotId && n.type !== "virtual"
  );
  const CLUSTER_THRESH = 20; // ~20 grid units
  const clusters = detectGeoClusters(nonDepotReal, CLUSTER_THRESH);

  for (const [pivotId, memberIds] of clusters) {
    const safe = memberIds.filter(id => id !== depotId);
    if (safe.length < 2) continue;
    // Skip if all already replaced
    const stillPresent = safe.filter(id => current.nodes.some(n => n.id === id));
    if (stillPresent.length < 2) continue;

    const macroId = `CLUSTER_${pivotId}`;
    const { x, y } = getCentroid(stillPresent, current.nodes);
    const macro: Node = { id: macroId, x, y, type: "virtual" };
    const gadget: Gadget = {
      type: "denseCluster",
      id: `g_cluster_${counter++}`,
      replacedNodeIds: stillPresent,
      macroNode: macro,
      description: `Dense cluster of ${stillPresent.length} stops near (${x.toFixed(1)},${y.toFixed(1)})`,
    };
    gadgets.push(gadget);
    patterns.push({
      type: "denseCluster",
      id: gadget.id,
      involvedNodeIds: stillPresent,
      macroNodeId: macroId,
      description: `Dense Minor: ${stillPresent.length} stops within ${CLUSTER_THRESH} units of ${pivotId}. Micro-routed internally.`,
    });
    current = applyGadget(current, gadget);
    logs.push(`[Stage 2] Dense cluster near ${pivotId} (${stillPresent.length} stops) → ${macroId}`);
  }

  return { compressedGraph: current, gadgets, patterns };
}

// ─────────────────────────────────────────────────────────
// Stage 3: Recursive Refinement
// ─────────────────────────────────────────────────────────

function spectralPartition(
  nodes: Node[],
  depotId: string,
  ctx: TrafficContext
): { left: Node[]; right: Node[] } {
  const depot = nodes.find(n => n.id === depotId) ?? nodes[0];
  const sorted = [...nodes].sort((a, b) => {
    return (
      getEdgeCost(depot, a, ctx).composite -
      getEdgeCost(depot, b, ctx).composite
    );
  });
  const mid = Math.ceil(sorted.length / 2);
  return { left: sorted.slice(0, mid), right: sorted.slice(mid) };
}

function recursiveSolve(
  graph: InteractionGraph,
  depotId: string,
  ctx: TrafficContext,
  depth: number,
  maxDepth: number
): string[] {
  const { nodes } = graph;
  if (nodes.length === 0) return [];
  if (nodes.length <= 8 || depth >= maxDepth) {
    const nodeMap = new Map(nodes.map(n => [n.id, n]));
    const startId = nodes.some(n => n.id === depotId) ? depotId : nodes[0].id;
    const greedy = nearestNeighbor(nodes, startId, ctx);
    return twoOpt(greedy, nodeMap, ctx);
  }

  const { left, right } = spectralPartition(nodes, depotId, ctx);
  if (right.length === 0) {
    const nodeMap = new Map(nodes.map(n => [n.id, n]));
    const greedy = nearestNeighbor(nodes, depotId, ctx);
    return twoOpt(greedy, nodeMap, ctx);
  }

  const leftDepot = left.some(n => n.id === depotId) ? depotId : left[0].id;
  const rightDepot = right[0].id;

  const leftPath = recursiveSolve(
    { nodes: left, constraints: graph.constraints },
    leftDepot,
    ctx,
    depth + 1,
    maxDepth
  );
  const rightPath = recursiveSolve(
    { nodes: right, constraints: graph.constraints },
    rightDepot,
    ctx,
    depth + 1,
    maxDepth
  );

  const stitched = [...leftPath];
  const skip = rightPath[0] === leftPath[leftPath.length - 1] ? 1 : 0;
  stitched.push(...rightPath.slice(skip));
  return stitched;
}

// ─────────────────────────────────────────────────────────
// Gadget expansion: macro-nodes → original stops (micro-routed)
// ─────────────────────────────────────────────────────────

function expandGadgets(
  macroPath: string[],
  gadgets: Gadget[],
  allOriginalNodes: Node[],
  ctx: TrafficContext
): string[] {
  const result: string[] = [];

  for (const id of macroPath) {
    const gadget = gadgets.find(g => g.macroNode.id === id);
    if (!gadget) {
      result.push(id);
      continue;
    }
    const members = allOriginalNodes.filter(n =>
      gadget.replacedNodeIds.includes(n.id)
    );
    if (members.length === 0) continue;
    if (members.length === 1) {
      result.push(members[0].id);
      continue;
    }
    // Micro-route internally
    const nodeMap = new Map(members.map(n => [n.id, n]));
    const local = nearestNeighbor(members, members[0].id, ctx);
    const optimized = twoOpt(local, nodeMap, ctx);
    // Drop the return-to-depot at end of local sub-tour (it's an intermediate segment)
    result.push(...optimized.slice(0, -1));
  }

  return result;
}

// ─────────────────────────────────────────────────────────
// Main Engine Entry Point
// ─────────────────────────────────────────────────────────

export function runReasonRouteEngine(input: EngineInput): EngineResult {
  const logs: string[] = ["[Engine] GNOPT Spatial Reasoning Engine v4.0 — Start."];
  const ctx: TrafficContext = {
    heatmap: input.trafficHeatmap,
    weightTime: input.preferences?.weightTime ?? 0.5,
    weightEmissions: input.preferences?.weightEmissions ?? 0.3,
    weightDistance: input.preferences?.weightDistance ?? 0.2,
  };

  const allNodes = input.nodes;
  const constraints = input.constraints ?? [];
  const depotId = input.depotId;
  const nodeMap = new Map(allNodes.map(n => [n.id, n]));

  // Stage 0
  logs.push("[Stage 0] Normalizing constraints to arity ≤ 3.");
  const normGraph = normalizeConstraints(allNodes, constraints, logs);
  logs.push(
    `[Stage 0] ${allNodes.length} nodes | ${normGraph.constraints.length} normalized constraints.`
  );

  // Stage 1 — Baseline
  logs.push("[Stage 1] Composite-cost nearest-neighbor baseline + 2-opt refinement.");
  const rawGreedy = nearestNeighbor(allNodes, depotId, ctx);
  const baselinePath = twoOpt(rawGreedy, nodeMap, ctx);
  const baselineMetrics = evaluatePath(baselinePath, nodeMap, ctx);
  logs.push(
    `[Stage 1] Baseline → ${baselineMetrics.distance.toFixed(2)} km | ` +
    `${baselineMetrics.time.toFixed(1)} min | ` +
    `${(baselineMetrics.emissions / 1000).toFixed(3)} kg CO₂ | ` +
    `Score: ${baselineMetrics.composite.toFixed(2)}`
  );

  // Stage 2 — Compression
  logs.push("[Stage 2] Preprocessor P — Hub / Sunflower / Dense Cluster detection.");
  const { compressedGraph, gadgets, patterns } = runPreprocessorP(
    normGraph,
    depotId,
    logs
  );
  const pctReduced = (
    (1 - compressedGraph.nodes.length / Math.max(allNodes.length, 1)) * 100
  ).toFixed(1);
  logs.push(
    `[Stage 2] ${allNodes.length} → ${compressedGraph.nodes.length} nodes (${pctReduced}% compressed). ` +
    `${patterns.length} pattern(s) found.`
  );

  // Stage 3 — Recursive Solve
  const maxDepth = Math.max(1, Math.ceil(Math.log2(Math.max(compressedGraph.nodes.length, 2))));
  logs.push(`[Stage 3] Recursive solve on compressed graph (max depth ${maxDepth}).`);
  const macroCandidatePath = recursiveSolve(
    compressedGraph,
    depotId,
    ctx,
    0,
    maxDepth
  );

  // Expand macro-nodes back to real stops
  const expandedPath = expandGadgets(macroCandidatePath, gadgets, allNodes, ctx);

  // Ensure depot is at start
  const finalPath =
    expandedPath.length > 0 && expandedPath[0] === depotId
      ? expandedPath
      : [depotId, ...expandedPath.filter(id => id !== depotId)];

  // Evaluate on ORIGINAL coordinates — same nodeMap as baseline
  const optimizedMetrics = evaluatePath(finalPath, nodeMap, ctx);
  logs.push(
    `[Stage 3] Optimized → ${optimizedMetrics.distance.toFixed(2)} km | ` +
    `${optimizedMetrics.time.toFixed(1)} min | ` +
    `${(optimizedMetrics.emissions / 1000).toFixed(3)} kg CO₂ | ` +
    `Score: ${optimizedMetrics.composite.toFixed(2)}`
  );

  // Improvement percentages
  const pct = (before: number, after: number) =>
    before > 0 ? +((( before - after) / before) * 100).toFixed(1) : 0;

  const improvement = {
    distancePct: pct(baselineMetrics.distance, optimizedMetrics.distance),
    timePct: pct(baselineMetrics.time, optimizedMetrics.time),
    emissionsPct: pct(baselineMetrics.emissions, optimizedMetrics.emissions),
  };

  const arrow = (v: number) => (v > 0 ? `↓${v}%` : v < 0 ? `↑${Math.abs(v)}%` : `→0%`);
  logs.push(
    `[Result] Distance ${arrow(improvement.distancePct)} | ` +
    `Time ${arrow(improvement.timePct)} | ` +
    `Emissions ${arrow(improvement.emissionsPct)}`
  );

  // Log disruption zones if present
  for (const zone of input.disruptionZones ?? []) {
    logs.push(
      `[Context] Disruption zone: "${zone.description}" at (${zone.x}, ${zone.y}), r=${zone.radius}.`
    );
  }

  return {
    baselinePath,
    optimizedPath: finalPath,
    metrics: { baseline: baselineMetrics, optimized: optimizedMetrics, improvement },
    detectedPatterns: patterns,
    engineLogs: logs,
    recursionDepth: maxDepth,
  };
}

// ─────────────────────────────────────────────────────────
// Step 1: Parse stops text → deterministic Node[]
// ─────────────────────────────────────────────────────────

function parseStopsToNodes(stopsInput: string): { nodes: Node[]; depotId: string } {
  const lines = stopsInput
    .split("\n")
    .map(l => l.trim())
    .filter(Boolean);

  if (lines.length === 0) throw new Error("No stops provided.");

  const nodes: Node[] = lines.map((line, i) => {
    const parts = line.split(",").map(p => p.trim());
    const label = parts[0];
    const id = `node_${i}`;

    let x: number, y: number;

    if (parts.length >= 3 && !isNaN(Number(parts[1])) && !isNaN(Number(parts[2]))) {
      x = Math.min(95, Math.max(5, Number(parts[1])));
      y = Math.min(95, Math.max(5, Number(parts[2])));
    } else {
      // Spiral layout — deterministic spread on 100×100 grid
      const angle = (i / Math.max(lines.length, 1)) * 2 * Math.PI;
      const radius = i === 0 ? 0 : 20 + (i % 4) * 12;
      x = +(50 + radius * Math.cos(angle)).toFixed(1);
      y = +(50 + radius * Math.sin(angle)).toFixed(1);
    }

    return {
      id,
      x,
      y,
      type: i === 0 ? "depot" : "stop",
      label,
    } as Node & { label: string };
  });

  return { nodes, depotId: nodes[0].id };
}

  try {

export const optimizeRoute = async (
  stopsInput: string,
  contextImageBase64?: string,
  contextImageMimeType?: string,
  preferences?: {
    weightTime?: number;
    weightEmissions?: number;
    weightDistance?: number;
  }
): Promise<OptimizationResult> => {

  // 1. Parse stops deterministically
  const { nodes, depotId } = parseStopsToNodes(stopsInput);

  // 2. Extract disruptions from image (Gemini visual role)
  let disruptions: DisruptionZone[] = [];
  let heatmapForEngine: number[][] | undefined;

  if (contextImageBase64 && contextImageMimeType) {
    disruptions = await extractDisruptionsFromImage(contextImageBase64, contextImageMimeType);

    // Convert disruptions → heatmap grid for the engine
    if (disruptions.length > 0) {
      const grid: number[][] = Array.from({ length: 100 }, () =>
        new Array(100).fill(0)
      );
      for (const zone of disruptions) {
        const cx = Math.round(Math.min(99, Math.max(0, zone.x)));
        const cy = Math.round(Math.min(99, Math.max(0, zone.y)));
        const r = Math.max(1, Math.min(30, zone.radius));
        for (let dx = -r; dx <= r; dx++) {
          for (let dy = -r; dy <= r; dy++) {
            const nx = cx + dx, ny = cy + dy;
            if (nx >= 0 && nx < 100 && ny >= 0 && ny < 100) {
              const severity = Math.max(0, 1 - Math.sqrt(dx * dx + dy * dy) / r) * 0.9;
              grid[nx][ny] = Math.max(grid[nx][ny], severity);
            }
          }
        }
      }
      heatmapForEngine = grid;
    }
  }

  // 3. Run deterministic GNOPT math engine
  const engineInput: EngineInput = {
    nodes,
    depotId,
    trafficHeatmap: heatmapForEngine,
    disruptionZones: disruptions,
    preferences: {
      weightTime: preferences?.weightTime ?? 0.5,
      weightEmissions: preferences?.weightEmissions ?? 0.3,
      weightDistance: preferences?.weightDistance ?? 0.2,
    },
  };

  const engineResult = runReasonRouteEngine(engineInput);

  // 4. Gemini explains the engine's decisions (no math, only narrative)
  const reasoningLog = await generateReasoningLog(nodes, engineResult, disruptions);

  // 5. Return — emissions converted grams → kg for UI
  const toKg = (g: number) => +(g / 1000).toFixed(3);

  return {
    nodes,
    baselinePath: engineResult.baselinePath,
    optimizedPath: engineResult.optimizedPath,
    disruptions,
    logs: reasoningLog,
    engineLogs: engineResult.engineLogs,
    metrics: {
      baseline: {
        distance: +engineResult.metrics.baseline.distance.toFixed(2),
        time: +engineResult.metrics.baseline.time.toFixed(1),
        emissions: toKg(engineResult.metrics.baseline.emissions),
      },
      optimized: {
        distance: +engineResult.metrics.optimized.distance.toFixed(2),
        time: +engineResult.metrics.optimized.time.toFixed(1),
        emissions: toKg(engineResult.metrics.optimized.emissions),
      },
      improvement: engineResult.metrics.improvement,
    },
    detectedPatterns: engineResult.detectedPatterns,
  };
};




// baslineSolver.ts
// Shared math primitives for the GNOPT Spatial Reasoning Engine.
// Exports all types and functions consumed by reasonRouteEngine.ts

export const SCALE_KM = 1.0;

export type Node = {
  id: string;
  x: number;
  y: number;
  type?: string;
  label?: string;
  constraints?: any[];
};

export type TrafficContext = {
  heatmap?: number[][];
  weightTime: number;
  //disruptionZones?: { x: number; y: number; radius: number; description: string }[];
  weightEmissions: number;
  weightDistance: number;
};

export type PathMetrics = {
  distance:  number;
  time:      number;
  emissions: number;
  composite: number;
};

const BASE_SPEED_KMH      = 40;
const HIGHWAY_SPEED_KMH   = 90;
const HIGHWAY_THRESH_KM   = 5;
const EMISSION_BASE_G_KM  = 120;
const EMISSION_IDLE_G_KM  = 250;
const IDLE_TRAFFIC_THRESH = 0.4;

export function euclidean(a: Node, b: Node): number {
  const dx = a.x - b.x;
  const dy = a.y - b.y;
  return Math.sqrt(dx * dx + dy * dy) * SCALE_KM;
}

export function getEdgeCost(a: Node, b: Node, ctx: TrafficContext): PathMetrics {
  const distKm = euclidean(a, b);

  let trafficSeverity = 0;
  if (ctx.heatmap && ctx.heatmap.length > 0) {
    const rows = ctx.heatmap.length;
    const cols = ctx.heatmap[0]?.length ?? 0;
    if (rows > 0 && cols > 0) {
      const midX = (a.x + b.x) / 2;
      const midY = (a.y + b.y) / 2;
      const row = Math.min(rows - 1, Math.max(0, Math.floor((midY / 100) * rows)));
      const col = Math.min(cols - 1, Math.max(0, Math.floor((midX / 100) * cols)));
      trafficSeverity = Math.max(0, Math.min(1, ctx.heatmap[row]?.[col] ?? 0));
    }
  }

  const trafficDelay  = 1.0 + trafficSeverity * 3.0;
  const isHighway     = distKm > HIGHWAY_THRESH_KM;
  const rawSpeed      = isHighway ? HIGHWAY_SPEED_KMH : BASE_SPEED_KMH;
  const effectiveSpd  = rawSpeed / trafficDelay;
  const timeMins      = distKm > 0 ? (distKm / effectiveSpd) * 60 : 0;
  const emissionRate  = trafficSeverity > IDLE_TRAFFIC_THRESH ? EMISSION_IDLE_G_KM : EMISSION_BASE_G_KM;
  const emissions     = distKm * emissionRate;

  const composite =
    timeMins              * ctx.weightTime      +
    (emissions / 100)     * ctx.weightEmissions +
    distKm                * ctx.weightDistance;

  return { distance: distKm, time: timeMins, emissions, composite };
}

export function nearestNeighbor(
  nodes: Node[],
  startId: string,
  ctx: TrafficContext
): string[] {
  if (nodes.length === 0) return [];
  const unvisited = new Map(nodes.map(n => [n.id, n]));
  const start = unvisited.get(startId);
  if (!start) throw new Error(`Start node '${startId}' not found`);

  const path: string[] = [startId];
  unvisited.delete(startId);
  let current = start;

  while (unvisited.size > 0) {
    let best: Node | null = null;
    let bestCost = Infinity;
    for (const candidate of unvisited.values()) {
      const cost = getEdgeCost(current, candidate, ctx).composite;
      if (cost < bestCost) { bestCost = cost; best = candidate; }
    }
    if (!best) break;
    path.push(best.id);
    unvisited.delete(best.id);
    current = best;
  }

  path.push(startId);
  return path;
}

export function twoOpt(
  path: string[],
  nodeMap: Map<string, Node>,
  ctx: TrafficContext
): string[] {
  let best = [...path];
  let improved = true;
  let iterations = 0;

  while (improved && iterations < 100) {
    improved = false;
    iterations++;
    for (let i = 1; i < best.length - 2; i++) {
      for (let k = i + 1; k < best.length - 1; k++) {
        const a = nodeMap.get(best[i - 1]);
        const b = nodeMap.get(best[i]);
        const c = nodeMap.get(best[k]);
        const d = nodeMap.get(best[k + 1]);
        if (!a || !b || !c || !d) continue;
        const cur = getEdgeCost(a, b, ctx).composite + getEdgeCost(c, d, ctx).composite;
        const nxt = getEdgeCost(a, c, ctx).composite + getEdgeCost(b, d, ctx).composite;
        if (nxt < cur - 1e-9) {
          best = [...best.slice(0, i), ...best.slice(i, k + 1).reverse(), ...best.slice(k + 1)];
          improved = true;
        }
      }
    }
  }
  return best;
}

export function evaluatePath(
  path: string[],
  nodeMap: Map<string, Node>,
  ctx: TrafficContext
): PathMetrics {
  const total: PathMetrics = { distance: 0, time: 0, emissions: 0, composite: 0 };
  for (let i = 0; i < path.length - 1; i++) {
    const a = nodeMap.get(path[i]);
    const b = nodeMap.get(path[i + 1]);
    if (a && b) {
      const e = getEdgeCost(a, b, ctx);
      total.distance  += e.distance;
      total.time      += e.time;
      total.emissions += e.emissions;
      total.composite += e.composite;
    }
  }
  return total;
}

export function routeDistance(path: string[], nodeMap: Map<string, Node>): number {
  let total = 0;
  for (let i = 0; i < path.length - 1; i++) {
    const a = nodeMap.get(path[i]);
    const b = nodeMap.get(path[i + 1]);
    if (a && b) total += euclidean(a, b);
  }
  return total;
}

// services/baslineSolver.ts
// Shared math primitives for the GNOPT Spatial Reasoning Engine.

export const SCALE_KM = 1.0;

export type Node = {
  id: string;
  x: number;
  y: number;
  type?: string;
  label?: string;
  constraints?: any[];
};

export type TrafficContext = {
  heatmap?: number[][];
  disruptionZones?: { x: number; y: number; radius: number; description: string }[];
  weightTime: number;
  weightEmissions: number;
  weightDistance: number;
};

export type PathMetrics = {
  distance:  number;
  time:      number;
  emissions: number;
  composite: number;
};

// PHYSICS CONSTANTS
const BASE_SPEED_KMH      = 40;
const HIGHWAY_SPEED_KMH   = 90;
const HIGHWAY_THRESH_KM   = 5;
const EMISSION_BASE_G_KM  = 120;
const EMISSION_IDLE_G_KM  = 400; // High penalty for gridlock
const IDLE_TRAFFIC_THRESH = 0.4;

// ---------------------------------------------------------
// 1. CORE MATH
// ---------------------------------------------------------

export function euclidean(a: Node, b: Node): number {
  const dx = a.x - b.x;
  const dy = a.y - b.y;
  return Math.sqrt(dx * dx + dy * dy) * SCALE_KM;
}

export function routeDistance(path: string[], nodeMap: Map<string, Node>): number {
  let total = 0;
  for (let i = 0; i < path.length - 1; i++) {
    const a = nodeMap.get(path[i]);
    const b = nodeMap.get(path[i + 1]);
    if (a && b) total += euclidean(a, b);
  }
  return total;
}

// ---------------------------------------------------------
// 2. THE "REAL WORLD" COST EVALUATOR
// ---------------------------------------------------------
// This calculates the TRUE cost (Time/Emissions) including Traffic & Fire.
// Both the Baseline and Optimized paths are JUDGED by this function.

export function getEdgeCost(a: Node, b: Node, ctx: TrafficContext): PathMetrics {
  const distKm = euclidean(a, b);

  // A. Scan for Traffic/Heatmap
  let trafficSeverity = 0;
  if (ctx.heatmap && ctx.heatmap.length > 0) {
    const getSev = (x: number, y: number) => {
       const rows = ctx.heatmap!.length;
       const cols = ctx.heatmap![0]?.length || 0;
       const r = Math.min(Math.max(Math.floor(x), 0), rows - 1);
       const c = Math.min(Math.max(Math.floor(y), 0), cols - 1);
       return ctx.heatmap![r]?.[c] ?? 0;
    };
    // Check multiple points to catch fire in the middle of a segment
    const s1 = getSev(a.x, a.y);
    const s2 = getSev(b.x, b.y);
    const s3 = getSev((a.x + b.x)/2, (a.y + b.y)/2);
    trafficSeverity = Math.max(s1, s2, s3);
  }

  // B. Scan for Disruption Zones
  if (ctx.disruptionZones) {
      const hit = ctx.disruptionZones.some(z => {
          const mx = (a.x + b.x)/2;
          const my = (a.y + b.y)/2;
          return Math.sqrt((mx-z.x)**2 + (my-z.y)**2) <= z.radius;
      });
      if (hit) trafficSeverity = 1.0;
  }

  // C. Calculate Physics
  const trafficDelay  = 1.0 + (trafficSeverity * trafficSeverity * 10);
  const isHighway     = distKm > HIGHWAY_THRESH_KM;
  const rawSpeed      = isHighway ? HIGHWAY_SPEED_KMH : BASE_SPEED_KMH;
  const effectiveSpd  = Math.max(1, rawSpeed / trafficDelay); 
  
  const timeMins      = distKm > 0 ? (distKm / effectiveSpd) * 60 : 0;
  const emissionRate  = trafficSeverity > IDLE_TRAFFIC_THRESH ? EMISSION_IDLE_G_KM : EMISSION_BASE_G_KM;
  const emissions     = distKm * emissionRate;

  const composite =
    timeMins              * ctx.weightTime      +
    (emissions / 100)     * ctx.weightEmissions +
    distKm                * ctx.weightDistance;

  return { distance: distKm, time: timeMins, emissions, composite };
}

// ---------------------------------------------------------
// 3. STANDARD SOLVERS (The Baseline)
// ---------------------------------------------------------
// These represent a "Standard GPS" that optimizes for DISTANCE only.
// They purposefully IGNORE the Heatmap/Fire during decision making.
// This is critical to prove that your Context-Aware Engine is smarter.

export function nearestNeighbor(
  nodes: Node[],
  startId: string,
  ctx: TrafficContext
): string[] {
  if (nodes.length === 0) return [];
  const unvisited = new Set(nodes.map(n => n.id));
  const nodeMap = new Map(nodes.map(n => [n.id, n]));
  const path: string[] = [startId];
  unvisited.delete(startId);
  
  let currentId = startId;

  while (unvisited.size > 0) {
    let bestId = "";
    let minDist = Infinity;
    const current = nodeMap.get(currentId)!;

    for (const candidateId of unvisited) {
      const candidate = nodeMap.get(candidateId)!;
      // STANDARD LOGIC: Shortest Distance
      const d = euclidean(current, candidate);
      if (d < minDist) {
        minDist = d;
        bestId = candidateId;
      }
    }

    if (bestId) {
      path.push(bestId);
      unvisited.delete(bestId);
      currentId = bestId;
    } else {
      break;
    }
  }
  
  if (path[0] !== path[path.length-1]) path.push(path[0]);
  return path;
}

export function twoOpt(
  path: string[],
  nodeMap: Map<string, Node>,
  ctx: TrafficContext
): string[] {
  let best = [...path];
  let improved = true;
  let iterations = 0;

  while (improved && iterations < 50) {
    improved = false;
    iterations++;
    for (let i = 1; i < best.length - 2; i++) {
      for (let k = i + 1; k < best.length - 1; k++) {
        const a = nodeMap.get(best[i - 1])!;
        const b = nodeMap.get(best[i])!;
        const c = nodeMap.get(best[k])!;
        const d = nodeMap.get(best[k + 1])!;
        
        // STANDARD LOGIC: Shortest Distance
        const curDist = euclidean(a, b) + euclidean(c, d);
        const newDist = euclidean(a, c) + euclidean(b, d);

        if (newDist < curDist) {
          best = [...best.slice(0, i), ...best.slice(i, k + 1).reverse(), ...best.slice(k + 1)];
          improved = true;
        }
      }
    }
  }
  return best;
}

// ---------------------------------------------------------
// 4. EVALUATOR (The Judge)
// ---------------------------------------------------------
// This function applies the REAL-WORLD cost to the chosen path.
// If the Baseline chose a short path through a fire, this function
// will report a massive Time penalty.

export function evaluatePath(
  path: string[],
  nodeMap: Map<string, Node>,
  ctx: TrafficContext
): PathMetrics {
  const total: PathMetrics = { distance: 0, time: 0, emissions: 0, composite: 0 };
  for (let i = 0; i < path.length - 1; i++) {
    const a = nodeMap.get(path[i]);
    const b = nodeMap.get(path[i + 1]);
    if (a && b) {
      // THE JUDGE uses getEdgeCost (Smart)
      const e = getEdgeCost(a, b, ctx);
      total.distance  += e.distance;
      total.time      += e.time;
      total.emissions += e.emissions;
      total.composite += e.composite;
    }
  }
  return total;
}
